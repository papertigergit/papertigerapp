  `item_id` int(11) DEFAULT NULL,
  `action` int(11) DEFAULT NULL,
  `action_time` int(11) DEFAULT NULL,
  `by_who` int(11) DEFAULT NULL,

  `name` varchar(255) DEFAULT NULL,
  `keywords` text,

  `location_id` int(11) NOT NULL,
